import telebot
import os
import time
from flask import Flask
from threading import Thread

TOKEN = os.getenv("TELEGRAM_TOKEN")  # توکن از متغیر محیطی خونده میشه
bot = telebot.TeleBot(TOKEN)

# Flask app برای نگه داشتن پورت باز
app = Flask(__name__)

@app.route('/')
def home():
    return "ربات در حال اجراست!"

@app.route('/health')
def health():
    return "OK"

# هندلر سلام
@bot.message_handler(commands=['start'])
def send_welcome(message):
    bot.reply_to(message, "سلام! به ربات خوش اومدی. به من «سلام» بگو تا جواب بدم.")

@bot.message_handler(func=lambda message: message.text and message.text.strip() == "سلام")
def say_hello(message):
    bot.reply_to(message, "سلام!")

@bot.message_handler(func=lambda message: True)
def other_messages(message):
    bot.reply_to(message, "فقط به «سلام» پاسخ می‌دم.")

# اجرای ربات در ترد جداگانه
def run_bot():
    bot.infinity_polling()

if __name__ == "__main__":
    # ربات رو در ترد جداگانه اجرا کن
    bot_thread = Thread(target=run_bot)
    bot_thread.start()
    # فلاسک رو برای نگهداری پورت باز اجرا کن
    port = int(os.environ.get("PORT", 5000))
    app.run(host='0.0.0.0', port=port)
